package com.mycompany.homework2;

public class Homework2 {

    public static void main(String[] args) {
        int r = 5; 

        for (int i = 1; i <= r; i++) {
            for (int j = 0; j < r - i; j++) {
                System.out.print(" ");
            }
            for (int k = 1; k <= i; k++) {
                System.out.print("*");
            }
            System.out.println();
        }
    }
}
